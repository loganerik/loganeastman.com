{{- /* Deterministic cover attributes for a column. Returns a dict. style and accent come from separate seeds so they combine freely; variant tunes stripe pitch/width; motif picks the bottom ornament. */ -}} {{- $p := . -}} {{- $cover := $p.Params.cover | default dict -}} {{- $styles := slice "lines" "waves" "steps" "rays" "arches" "scallops" -}} {{- $accents := slice "sage" "rose" "jade" "brown" "olive" -}} {{- $tl := len $p.Title -}} {{- $dl := len ($p.Description | default "") -}} {{- $yd := $p.Date.YearDay -}} {{- $yr := $p.Date.Year -}} {{- $mo := int $p.Date.Month -}} {{- $dy := $p.Date.Day -}} {{- /* Auto style/accent rotate by the column's position in date order (oldest = 0), so consecutive columns never share a pattern or palette. If the previous column pinned the same value by hand, step past it. Front matter overrides always win. */ -}} {{- $cols := (where site.RegularPages "Section" "columns").ByDate -}} {{- $idx := 0 -}} {{- range $i, $e := $cols }}{{ if eq $e.RelPermalink $p.RelPermalink }}{{ $idx = $i }}{{ end }}{{ end -}} {{- $prevCover := dict -}} {{- if gt $idx 0 }}{{ $prevCover = (index $cols (sub $idx 1)).Params.cover | default dict }}{{ end -}} {{- $variant := mod (add $dy (mul $tl 5)) 3 -}} {{- $motif := mod (add (add $yd $yr) $dl) 3 -}} {{- $style := $cover.style | default "" -}} {{- if not (in $styles $style) -}} {{- $style = index $styles (mod $idx (len $styles)) -}} {{- if eq $style ($prevCover.style | default "") }}{{ $style = index $styles (mod (add $idx 1) (len $styles)) }}{{ end -}} {{- end -}} {{- $accent := $cover.accent | default "" -}} {{- if not (in $accents $accent) -}} {{- $accent = index $accents (mod $idx (len $accents)) -}} {{- if eq $accent ($prevCover.accent | default "") }}{{ $accent = index $accents (mod (add $idx 1) (len $accents)) }}{{ end -}} {{- end -}} {{- $inks := dict "sage" "#39745b" "rose" "#bc7f82" "jade" "#39847a" "brown" "#8d6e58" "olive" "#768b60" -}} {{- $tints := dict "sage" "#edf3e8" "rose" "#f6ede9" "jade" "#e9f2ee" "brown" "#f3efe6" "olive" "#eff2e6" -}} {{- $deeps := dict "sage" "#214634" "rose" "#7e4a50" "jade" "#1f4f48" "brown" "#52392c" "olive" "#445335" -}} {{- /* Duotone shadow→highlight tables per accent (feComponentTransfer tableValues for R, G, B). */ -}} {{- $duos := dict "sage" (dict "r" "0.13 1" "g" "0.29 0.99" "b" "0.21 0.97") "rose" (dict "r" "0.36 1" "g" "0.18 0.96" "b" "0.21 0.95") "jade" (dict "r" "0.11 0.96" "g" "0.30 1" "b" "0.28 0.98") "brown" (dict "r" "0.29 1" "g" "0.20 0.97" "b" "0.14 0.92") "olive" (dict "r" "0.22 0.98" "g" "0.29 1" "b" "0.16 0.94") -}} {{- return (dict "duo" (index $duos $accent) "style" $style "accent" $accent "variant" $variant "motif" $motif "ink" (index $inks $accent) "tint" (index $tints $accent) "deep" (index $deeps $accent)) -}}
